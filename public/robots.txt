# AffixAI — robots policy.
#
# Auth-walled app routes are blocked because they 401 to crawlers and
# we don't want soft 404s clogging up the index.
#
# /login and /register are PUBLIC pages — allowed so Google can index
# them and surface "Log In" / "Sign Up Free" as sitelinks under the
# brand search result.
#
# /invite/:token (guest signing magic links) are blocked — one-time-use
# tokens that should never appear in search results.

User-agent: *
Allow: /
Allow: /login
Allow: /register
Disallow: /dashboard
Disallow: /documents
Disallow: /data-vault
Disallow: /signatures
Disallow: /passport-photo
Disallow: /billing
Disallow: /admin
Disallow: /referrals
Disallow: /settings
Disallow: /enterprise
Disallow: /invite/
Disallow: /forgot-password
Disallow: /reset-password
